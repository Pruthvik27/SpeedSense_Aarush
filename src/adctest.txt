#include <stdio.h>

#include "ch32fun.h"
#include "clock.h"
#include "uart.h"
#include "adc.h"

void GPIO_Analog_Init(void)
{
    RCC->APB2PCENR |= RCC_APB2Periph_GPIOA;

    // PA0 -> ADC CH0
    GPIOA->CFGLR &= ~(0xF << (0 * 4));

    // PA1 -> ADC CH1
    GPIOA->CFGLR &= ~(0xF << (1 * 4));

    // PA3 -> ADC CH3
    GPIOA->CFGLR &= ~(0xF << (3 * 4));

    // PA4 -> ADC CH4
    GPIOA->CFGLR &= ~(0xF << (4 * 4));
}

int main(void)
{
    Clock_init();
    Delay_Ms(100);

    Uart_Init();
    Delay_Ms(100);

    GPIO_Analog_Init();

    ADC_Init();

    ADC_AddChannel(0);
    ADC_AddChannel(1);
    ADC_AddChannel(3);
    ADC_AddChannel(4);

    ADC_SetSample(0, 7);
    ADC_SetSample(1, 7);
    ADC_SetSample(3, 7);
    ADC_SetSample(4, 7);

    ADC_Start();

    char buffer[128];

    while (1)
    {
        sprintf(buffer,
                "CH0=%4u  CH1=%4u  CH3=%4u  CH4=%4u\r\n",
                ADC_GetValue(0),
                ADC_GetValue(1),
                ADC_GetValue(3),
                ADC_GetValue(4));

        print_string(buffer);

        Delay_Ms(500);
    }
}
